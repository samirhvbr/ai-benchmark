    @Test
    void aStatementWithABadLineRecordsNothing() throws Exception {

        byte[] statement = ofx(new String[]{"F1", "20260303", "300.00", "PIX " + ourNumberOf(unitA101, "2026-02")},

        assertEquals(0, count("select count(*) from payment"));

    }
